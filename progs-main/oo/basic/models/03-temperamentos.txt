class Temperamento {
  id: Integer
  nome: String
  elemento: String
  focoProfissional: String
  pontos_fortes: list[PontoForte]
  desafios: list[Desafio]
  personalidades: list[Personalidade]
  areas: list[AreaIdeal]
}

class PontoForte {
  descricao: String
}

class Desafio {
  descricao: String
}

class Personalidade {
  nome: String
}

class AreaIdeal {
  nome: String
  profissoes: list[Profissao]
}

class Profissao {
  nome: String
}

Temperamento *-- PontoForte

AreaIdeal "1" -- "1..*" Profissao : contem >
