@startuml

class Cliente {
  + nome : str
  + data_nascimento : Date
  + cpf : str
}

class Prato {
  + nome : str
  + ingredientes : list[str]
  + modo_preparo : str
  + preco : float
}

class Pedido {
  + data_pedido : Date
  + percentual_desconto : float
  + cliente : Cliente
  + itens : list[ItemDoPedido]
  + valor_total() : float
}

class ItemDoPedido {
  + prato: Prato
  + valor_prato: float
  + quantidade: int
}

Cliente  - Pedido 
Pedido -  ItemDoPedido
ItemDoPedido -- Prato


@enduml
